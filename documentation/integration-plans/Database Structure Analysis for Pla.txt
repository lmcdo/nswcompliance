● Specific Integration Mappings

  NSW API → Database Lookups:

  1. Zone-Based Provision Retrieval
  SELECT provision_text, ref_number
  FROM regulatory_provisions
  WHERE zone LIKE '%R2%'
    AND document_id LIKE '%Inner_West%'
    AND provision_text IS NOT NULL

  2. Heritage Area Provisions
  SELECT provision_text, control_type, value_text
  FROM regulatory_provisions rp
  JOIN development_controls dc ON rp.id = dc.provision_id
  WHERE rp.provision_text LIKE '%Haberfield%'
    OR rp.provision_text LIKE '%Conservation Area%'

  3. SEPP Override Logic
  SELECT sepp_provision_id, lep_clause_reference,
  override_type
  FROM sepp_lep_overrides
  WHERE lep_clause_reference IN ('4.3', '4.4')  --
  Height/FSR clauses

  4. Development Permissibility
  SELECT development_type, permission_status, conditions
  FROM development_permissions
  WHERE zone = 'R2'
    AND development_type LIKE '%dual_occupancy%'
